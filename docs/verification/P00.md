`data/*.csv` files, R2 inventory, and two Agent order files:

```powershell
$python = 'C:\Users\jaydo_0v7vk2o\.cache\codex-runtimes\codex-primary-runtime\dependencies\python\python.exe'
@'
from pathlib import Path
import csv
import pandas as pd

paths = sorted(Path("data").glob("*.csv")) + [
    Path("r2_objects.csv"),
    Path("agents/agent_0/orders/upcoming.csv"),
    Path("agents/agent_0/orders/previous.csv"),
]

for path in paths:
    with path.open("r", encoding="utf-8-sig", newline="") as source:
        header = next(csv.reader(source), [])
    duplicate_headers = sorted({
        name for name in header if header.count(name) > 1
    })
    frame = pd.read_csv(path, low_memory=False)
    columns = list(frame.columns)
    key = []
    for candidate in (
        ["date", "ticker"],
        ["date", "series_id"],
        ["date", "instrument_id"],
        ["observation_date", "source", "series_id", "maturity"],
        ["observation_date", "source", "instrument_id"],
        ["order_ref"],
        ["key"],
        ["date"],
    ):
        if all(column in columns for column in candidate):
            key = candidate
            break
    duplicate_keys = int(frame.duplicated(key).sum()) if key else None
    date_column = next((
        column for column in
        ["date", "observation_date", "activate_at", "last_modified"]
        if column in columns
    ), None)
    date_min = date_max = ""
    if date_column and len(frame):
        parsed = pd.to_datetime(
            frame[date_column], errors="coerce", utc=True
        )
        if parsed.notna().any():
            date_min = parsed.min().isoformat()
            date_max = parsed.max().isoformat()
    print(
        path.as_posix(),
        path.stat().st_size,
        len(frame),
        len(header),
        duplicate_headers,
        key,
        duplicate_keys,
        date_column,
        date_min,
        date_max,
        "|".join(header),
        sep="\t",
    )
'@ | & $python -
```

### Reproducible Eris-cache audit

```powershell
$python = 'C:\Users\jaydo_0v7vk2o\.cache\codex-runtimes\codex-primary-runtime\dependencies\python\python.exe'
@'
from pathlib import Path
from collections import Counter
import csv
import re

files = sorted(
    Path("data/cache/eris_sofr_settlements_v3").glob("*.csv")
)
schemas = Counter()
row_counts = []
duplicate_header_files = 0
exact_duplicate_files = 0
exact_duplicate_rows = 0
candidate_duplicate_files = 0
candidate_duplicate_rows = 0
filename_dates = []

for path in files:
    with path.open(
        "r", encoding="utf-8-sig", errors="replace", newline=""
    ) as source:
        reader = csv.reader(source)
        header = next(reader, [])
        schemas[tuple(header)] += 1
        if len(header) != len(set(header)):
            duplicate_header_files += 1
        evaluation_index = header.index("EvaluationDate")
        symbol_index = header.index("Symbol")
        seen_rows = set()
        seen_candidates = set()
        row_count = file_exact = file_candidate = 0
        for row in reader:
            row_count += 1
            row_tuple = tuple(row)
            if row_tuple in seen_rows:
                file_exact += 1
            else:
                seen_rows.add(row_tuple)
            candidate = (row[evaluation_index], row[symbol_index])
            if candidate in seen_candidates:
                file_candidate += 1
            else:
                seen_candidates.add(candidate)
        row_counts.append(row_count)
        if file_exact:
            exact_duplicate_files += 1
            exact_duplicate_rows += file_exact
        if file_candidate:
            candidate_duplicate_files += 1
            candidate_duplicate_rows += file_candidate
    match = re.search(r"(20\d{6})", path.name)
    if match:
        filename_dates.append(match.group(1))

print(
    len(files),
    sum(row_counts),
    min(row_counts),
    max(row_counts),
    sum(count == 0 for count in row_counts),
    len(schemas),
    duplicate_header_files,
    exact_duplicate_files,
    exact_duplicate_rows,
    candidate_duplicate_files,
    candidate_duplicate_rows,
    min(filename_dates),
    max(filename_dates),
)
for header, count in schemas.most_common():
    print(count, len(header), "|".join(header), sep="\t")
'@ | & $python -
```
